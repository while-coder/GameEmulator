project(gameemulator LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)

# Offline, hash-pinned source inputs; never download a moving nightly at build time.
function(unpack archive sha folder)
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/vendor/${archive}" actual)
  if(NOT actual STREQUAL sha)
    message(FATAL_ERROR "Source checksum mismatch: ${archive}")
  endif()
  if(NOT EXISTS "${CMAKE_CURRENT_BINARY_DIR}/${folder}/.unpacked")
    file(ARCHIVE_EXTRACT INPUT "${CMAKE_CURRENT_SOURCE_DIR}/vendor/${archive}"
      DESTINATION "${CMAKE_CURRENT_BINARY_DIR}")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${folder}/.unpacked" "${actual}")
  endif()
endfunction()
unpack(fceumm-source-e0f3086.tar.gz 7a81aa960e0424f736ceaf3221b93663f7ae0b0a68483ac17f8eb0c4afff28f7 libretro-fceumm-e0f3086a8104f8fbbe49a2615b10401504a9533f)
unpack(oboe-1.9.3.tar.gz 9d2486b74bd396d9d9112625077d5eb656fd6942392dc25ebf222b184ff4eb61 oboe-1.9.3)
set(core "${CMAKE_CURRENT_BINARY_DIR}/libretro-fceumm-e0f3086a8104f8fbbe49a2615b10401504a9533f/src")
set(common "${core}/drivers/libretro/libretro-common")
# Mirrors upstream Makefile.common: HAVE_NTSC=1, HAVE_HDPACK=0, RGB565.
file(GLOB boards "${core}/boards/*.c" "${core}/input/*.c")
set(core_sources ${boards} "${core}/ntsc/nes_ntsc.c"
  "${core}/drivers/libretro/libretro.c" "${core}/drivers/libretro/libretro_dipswitch.c")
foreach(name cart cheat crc32 fceu-endian fceu-memory fceu fds fds_apu file filter general input md5 nsf palette ppu sound state video vsuni ines unif x6502)
  list(APPEND core_sources "${core}/${name}.c")
endforeach()
foreach(name streams/memory_stream compat/compat_posix_string compat/compat_snprintf compat/compat_strcasestr compat/compat_strl compat/fopen_utf8 encodings/encoding_utf file/file_path file/file_path_io streams/file_stream streams/file_stream_transforms string/stdstring time/rtime vfs/vfs_implementation)
  list(APPEND core_sources "${common}/${name}.c")
endforeach()
add_library(fceumm SHARED ${core_sources})
target_include_directories(fceumm PRIVATE "${core}" "${core}/input" "${core}/boards" "${core}/ntsc" "${core}/drivers/libretro" "${common}/include")
target_compile_definitions(fceumm PRIVATE __LIBRETRO__ PATH_MAX=1024 FCEU_VERSION_NUMERIC=9900 FRONTEND_SUPPORTS_RGB565 HAVE_NTSC_FILTER PSS_STYLE=1 GIT_VERSION=" e0f3086")
target_compile_options(fceumm PRIVATE -O3 -fvisibility=hidden)
target_link_options(fceumm PRIVATE "-Wl,--version-script=${core}/drivers/libretro/link.T")
target_link_libraries(fceumm PRIVATE m)
include(fbalpha.cmake)

add_library(gameemulator SHARED native_nes.cpp)
target_include_directories(gameemulator PRIVATE "${core}/drivers/libretro" "${common}/include")
target_compile_options(gameemulator PRIVATE -O3 -Wall -Wextra)
target_link_libraries(gameemulator PRIVATE oboe android log EGL GLESv2 dl)
foreach(target fceumm fbalpha2012 gameemulator)
  target_link_options(${target} PRIVATE "-Wl,-z,max-page-size=16384")
endforeach()
