cmake_minimum_required(VERSION 3.22)
project(console_core_check LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
function(unpack archive sha folder)
  set(vendor "${CMAKE_CURRENT_FUNCTION_LIST_DIR}/../../packages/android-native/android/src/main/cpp/vendor")
  file(SHA256 "${vendor}/${archive}" actual)
  if(NOT actual STREQUAL sha)
    message(FATAL_ERROR "Checksum mismatch: ${archive}")
  endif()
  if(NOT EXISTS "${CMAKE_CURRENT_BINARY_DIR}/${folder}")
    file(ARCHIVE_EXTRACT INPUT "${vendor}/${archive}" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}")
  endif()
endfunction()
